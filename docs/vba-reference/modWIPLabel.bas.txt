Option Explicit

Private slotsUsed As Integer

' Writes one WIP label into the next free slot on WIP_LABEL_SHEET_12UP.
' Does NOT print - call FlushWIPLabels once all labels for this batch
' (e.g. every bag from one Bag Entry save) have been queued.
Public Sub QueueWIPLabel( _
    PartMaterial As String, _
    BagNo As String, _
    WeightKg As Double, _
    Qty As Double, _
    ProdDate As Date)

    Const SHEET_NAME As String = "WIP_LABEL_SHEET_12UP"

    If slotsUsed = 0 Then
        Call SetupLabelSheet12Up(SHEET_NAME)
    ElseIf slotsUsed >= 12 Then
        Call FlushWIPLabels("NONE") ' sheet full mid-save - just clear and start a new page, no output prompt here
    End If

    slotsUsed = slotsUsed + 1

    Dim ws As Worksheet
    Set ws = Sheets(SHEET_NAME)
    If ws.FilterMode Then ws.ShowAllData

    Dim tl As Variant
    tl = SlotTopLeft(slotsUsed)
    Dim r0 As Long, c0 As Long
    r0 = tl(0): c0 = tl(1)

    ws.Cells(r0, c0).Value = PartMaterial
    ws.Cells(r0, c0).Font.Bold = True
    ws.Cells(r0, c0).Font.Size = 11

    ws.Cells(r0 + 2, c0).Value = "Bag Wt"
    ws.Cells(r0 + 2, c0 + 3).Value = Format(WeightKg, "0.000") & " Kg"

    ws.Cells(r0 + 4, c0).Value = "Aprox Qty"
    If Qty > 0 Then
        ws.Cells(r0 + 4, c0 + 3).Value = Format(Qty, "0") & " Nos"
    Else
        ws.Cells(r0 + 4, c0 + 3).Value = "-"
    End If

    ws.Cells(r0 + 6, c0).Value = "Prod Date"
    ws.Cells(r0 + 6, c0 + 3).Value = Format(ProdDate, "dd-mm-yyyy")

    ws.Cells(r0 + 8, c0).Value = "Batch No"
    ws.Cells(r0 + 8, c0 + 3).Value = BagNo

    ws.Cells(r0 + 11, c0).Value = "SRI HARI RUBBER PRODUCTS"
    ws.Cells(r0 + 11, c0).Font.Size = 7

    Dim qrCellAddr As String
    qrCellAddr = ws.Cells(r0, c0 + 5).Address

    Call GenerateQR(BagNo, SHEET_NAME, qrCellAddr, SHEET_NAME & "_QR_" & slotsUsed, 40)

End Sub

' Executes whatever labels have been queued (even a partial sheet),
' using an output choice already decided upfront via
' AskLabelOutputChoice - no prompting happens here. Then clears the
' sheet ready for the next batch. Call once after the last label in
' a save transaction has been queued.
Public Sub FlushWIPLabels(ByVal OutputChoice As String)

    Const SHEET_NAME As String = "WIP_LABEL_SHEET_12UP"

    If slotsUsed = 0 Then Exit Sub

    Dim ws As Worksheet
    Set ws = Sheets(SHEET_NAME)
    If ws.FilterMode Then ws.ShowAllData

    Call ExecuteLabelOutput(ws, "WIP_Batch_" & Format(Now, "yyyymmdd_hhmmss"), OutputChoice)

    Call ClearLabelSheet12Up(SHEET_NAME)
    slotsUsed = 0

End Sub


Public Sub PrintPartBagLabel( _
    BagNo As String, _
    BatchNo As String, _
    PartNo As String, _
    PartName As String, _
    WeightKg As Double, _
    Qty As Double, _
    ProdDate As Variant)

    Call QueueWIPLabel(PartNo, BagNo, WeightKg, Qty, CDate(ProdDate))

End Sub


Public Sub PrintRunnerBagLabel( _
    BagNo As String, _
    BatchNo As String, _
    MaterialName As String, _
    WeightKg As Double, _
    ProdDate As Variant)

    Call QueueWIPLabel(MaterialName, BagNo, WeightKg, 0, CDate(ProdDate))

End Sub
