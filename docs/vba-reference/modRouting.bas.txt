Option Explicit

Private Function FindPartRow(PartNo As String) As Long

    Dim ws As Worksheet
    Dim r As Long
    Dim LR As Long

    Set ws = Sheets(SH_PARTMASTER)
    If ws.FilterMode Then ws.ShowAllData

    LR = ws.Cells(ws.Rows.Count, "A").End(xlUp).Row

    For r = 2 To LR

        If Trim(UCase(ws.Cells(r, "A").Value)) = _
           Trim(UCase(PartNo)) Then

            FindPartRow = r
            Exit Function

        End If

    Next r

End Function


Public Function TrimRequired(PartNo As String) As Boolean

    Dim r As Long

    r = FindPartRow(PartNo)

    If r = 0 Then Exit Function

    TrimRequired = _
    (UCase(Sheets(SH_PARTMASTER).Cells(r, "D").Value) = "Y")

End Function
