Private Sub txtAcceptWeight_Change()

    Dim PW As Double

    PW = GetPartWeight(lblPart.Caption)

    If PW > 0 And val(txtAcceptWeight.Value) > 0 Then

        lblAcceptPieces.Caption = _
            Format(Round(val(txtAcceptWeight.Value) * 1000 / PW, 0), "#,##0")

    Else

        lblAcceptPieces.Caption = ""

    End If

    UpdateRemaining

End Sub

Private Sub chkSelectByPart_Click()

    If chkSelectByPart.Value Then

        ' Checked = Select by Barcode (direct entry/scan)
        cboPartNoFIFO.Visible = False
        cboPartNoFIFO.Value = ""
        txtBatch.Locked = False
        txtBatch.Value = ""

    Else

        ' Unchecked (default) = Select by Part - FIFO, since barcode
        ' labels aren't in use yet.
        cboPartNoFIFO.Visible = True
        txtBatch.Locked = True
        txtBatch.Value = ""

        Call PopulateFIFOPartDropdown(cboPartNoFIFO)

    End If

End Sub

Private Sub cboPartNoFIFO_AfterUpdate()

    If cboPartNoFIFO.Value = "" Then Exit Sub

    Dim foundBag As String
    foundBag = GetFIFOBagForInspection(cboPartNoFIFO.Value)

    If foundBag = "" Then

        MsgBox "No bags currently ready for Inspection for Part '" & _
               cboPartNoFIFO.Value & "'.", vbInformation, "No Bags Available"

        txtBatch.Value = ""
        Exit Sub

    End If

    txtBatch.Value = foundBag
    Call txtBatch_AfterUpdate

End Sub

Private Sub txtBatch_AfterUpdate()

    Dim BatchNo As String
    Dim ProdWt As Double
    Dim InspWt As Double
    Dim RemWt As Double
    Dim PW As Double

    BatchNo = Trim(txtBatch.Value)

    If BatchNo = "" Then Exit Sub

    If GetPartName(BatchNo) = "" Then

        MsgBox "Batch Number '" & BatchNo & _
               "' does not exist.", _
               vbExclamation, "Invalid Batch"

        txtBatch.Value = ""
        txtBatch.SetFocus

        Exit Sub

    End If

    If GetBagType(BatchNo) <> "PART" And GetBagType(BatchNo) <> "" Then

        MsgBox "'" & BatchNo & "' is a " & GetBagType(BatchNo) & " bag, not a Part bag." & vbCrLf & _
               "Only Part bags go through Inspection - Runner, Rejection and Lump entries are scrap/waste material.", _
               vbExclamation, "Not a Part Bag"

        txtBatch.Value = ""
        txtBatch.SetFocus

        Exit Sub

    End If

    If Not IsBatchInProductionLog(GetBatchNoForBag(BatchNo)) Then

        MsgBox "This bag's batch (" & GetBatchNoForBag(BatchNo) & _
               ") was not found in Production Log.", _
               vbExclamation, "Production Log Mismatch"

        txtBatch.Value = ""
        txtBatch.SetFocus

        Exit Sub

    End If

    If StageRank(GetBagStatus(BatchNo)) >= 2 Then

        If MsgBox( _
            "This bag is already at status '" & GetBagStatus(BatchNo) & _
            "'." & vbCrLf & vbCrLf & _
            "Add another Inspection entry anyway?", _
            vbQuestion + vbYesNo, "Bag Already Processed") = vbNo Then

            txtBatch.Value = ""
            txtBatch.SetFocus

            Exit Sub

        End If

    End If

    lblPart.Caption = GetPartName(BatchNo)

    Call LoadBagsForBatch(lstBagsInBatch, GetBatchNoForBag(BatchNo))

    If Not InspectionRequired(lblPart.Caption) Then

        If MsgBox( _
            "Part Master shows Inspection = N for '" & lblPart.Caption & _
            "' - this part normally skips Inspection." & vbCrLf & vbCrLf & _
            "Confirm: do you still want to record an Inspection entry for this bag?", _
            vbQuestion + vbYesNo, "Inspection Not Required") = vbNo Then

            txtBatch.Value = ""
            txtBatch.SetFocus

            Exit Sub

        End If

    End If

    ProdWt = Getproductionweight(BatchNo)
    InspWt = GetInspectedWeight(BatchNo)
    RemWt = RemainingInspectionWeight(BatchNo)

    lblProdWt.Caption = Format(ProdWt, "0.00")
    lblProcessedWt.Caption = Format(InspWt, "0.00")
    lblRemaining.Caption = Format(RemWt, "0.00")

    PW = GetPartWeight(lblPart.Caption)

    If PW > 0 Then

        lblOKQty.Caption = _
            Format(Round((ProdWt * 1000) / PW, 0), "#,##0")

    Else

        lblOKQty.Caption = ""

    End If

End Sub
Private Sub txtRejectWt_KeyDown(ByVal KeyCode As MSForms.ReturnInteger, ByVal Shift As Integer)

    If KeyCode = vbKeyReturn Then
        KeyCode = 0
        Call cmdAddReject_Click
    End If

End Sub

    chkSelectByPart.Caption = "Select by Barcode (uncheck = Select by Part)"
    chkSelectByPart.Value = False
    cboPartNoFIFO.Visible = True
    txtBatch.Locked = True
    Call PopulateFIFOPartDropdown(cboPartNoFIFO)

    LoadRejectReasons
    LoadOperators

    With lstRejects
        .ColumnCount = 3
        .ColumnWidths = "120;60;120"
    End With

    With lstBagsInBatch
        .ColumnCount = 4
        .ColumnWidths = "75 pt;60 pt;60 pt;65 pt"
        .Visible = False
    End With

End Sub
Private Sub lstRejects_DblClick(ByVal Cancel As MSForms.ReturnBoolean)

    If lstRejects.ListIndex < 0 Then Exit Sub

    If MsgBox("Remove this reject entry?" & vbCrLf & vbCrLf & _
              "Reason: " & lstRejects.List(lstRejects.ListIndex, 0) & vbCrLf & _
              "Weight: " & lstRejects.List(lstRejects.ListIndex, 1), _
              vbQuestion + vbYesNo, "Remove Reject Entry") = vbYes Then

        lstRejects.RemoveItem lstRejects.ListIndex
        UpdateRemaining

    End If

End Sub
Private Sub chkRework_Click()

    If chkRework.Value Then

        If Trim(txtBatch.Value) = "" Then

            Call PopulateReworkBagPicker(cboReworkBag, "Inspection")

            lstPendingRework.Visible = False
            cboReworkBag.Visible = True

            If cboReworkBag.ListCount = 0 Then
                MsgBox "No bags currently have outstanding rework for Inspection.", _
                       vbInformation, "No Pending Rework"
                chkRework.Value = False
                cboReworkBag.Visible = False
            End If

            Exit Sub

        End If

        Dim ReworkBaseWt As Double

        If IsTrimRequired(lblPart.Caption) Then
            ReworkBaseWt = GetTrimmedWeight(txtBatch.Value)
        Else
            ReworkBaseWt = Getproductionweight(txtBatch.Value)
        End If

        cboReworkBag.Visible = False
        lstPendingRework.Visible = True

        Call PopulateOutstandingReworkList( _
            lstPendingRework, txtBatch.Value, "Inspection", _
            ReworkBaseWt, "INSPECTION_LOG")

        If lstPendingRework.ListCount = 0 Then
            MsgBox "No outstanding rework found in the Rejection Log for this bag.", _
                   vbInformation, "No Pending Rework"
        End If

    Else

        lstPendingRework.Clear
        lstPendingRework.Visible = False
        cboReworkBag.Clear
        cboReworkBag.Visible = False

    End If

End Sub

' Fires when a bag is picked from the "no bag yet" rework dropdown.
' Reads the bag code from the hidden column (never from the visible
' text) so this can't break regardless of what's in Part Name/Reason.
Private Sub cboReworkBag_AfterUpdate()

    If cboReworkBag.ListIndex < 0 Then Exit Sub

    Dim BagCode As String
    BagCode = cboReworkBag.List(cboReworkBag.ListIndex, 1)

    txtBatch.Value = BagCode
    Call txtBatch_AfterUpdate

    Dim ReworkBaseWt2 As Double

    If IsTrimRequired(lblPart.Caption) Then
        ReworkBaseWt2 = GetTrimmedWeight(BagCode)
    Else
        ReworkBaseWt2 = Getproductionweight(BagCode)
    End If

    Dim pendingWt As Double
    pendingWt = GetTotalOutstandingRework(BagCode, "Inspection", ReworkBaseWt2, "INSPECTION_LOG")

    txtAcceptWeight.Value = Format(pendingWt, "0.000")

    cboReworkBag.Visible = False
    lstPendingRework.Visible = True

    Call PopulateOutstandingReworkList( _
        lstPendingRework, txtBatch.Value, "Inspection", _
        ReworkBaseWt2, "INSPECTION_LOG")

End Sub

Private Sub cmdAddReject_Click()

    Dim PW As Double
    Dim RejectPieces As Long

    If cboRejectReason.Value = "" Then

        MsgBox "Select Reject Reason"
        Exit Sub

    End If

    If txtRejectWt.Value = "" Then

        MsgBox "Enter Reject Weight"
        Exit Sub

    End If

    PW = GetPartWeight(lblPart.Caption)

    If PW > 0 Then

        RejectPieces = _
            Round((CDbl(txtRejectWt.Value) * 1000) / PW, 0)

    Else

        RejectPieces = 0

    End If

    lstRejects.AddItem

    lstRejects.List(lstRejects.ListCount - 1, 0) = _
        cboRejectReason.Value

    lstRejects.List(lstRejects.ListCount - 1, 1) = _
        txtRejectWt.Value

    lstRejects.List(lstRejects.ListCount - 1, 2) = _
        RejectPieces

    lstRejects.List(lstRejects.ListCount - 1, 3) = _
        GetRejectDisposition(cboRejectReason.Value)

    txtRejectWt.Value = ""

    cboRejectReason.Value = ""

    UpdateRemaining

End Sub
Private Sub SaveRejectLines()

    Dim ws As Worksheet
    Dim LR As Long
    Dim i As Long

    Set ws = Sheets("INSPECTION_REJECT_LOG")
    If ws.FilterMode Then ws.ShowAllData

    For i = 0 To lstRejects.ListCount - 1

        LR = ws.Cells(ws.Rows.Count, "A").End(xlUp).Row + 1

        ws.Cells(LR, "A").Value = Date
        ws.Cells(LR, "B").Value = txtBatch.Value
        ws.Cells(LR, "C").Value = lblPart.Caption
        ws.Cells(LR, "D").Value = cboOperator.Value

        ws.Cells(LR, "E").Value = lstRejects.List(i, 0) 'Reason

        ws.Cells(LR, "F").Value = lstRejects.List(i, 1) 'Weight

        ws.Cells(LR, "G").Value = lstRejects.List(i, 2) 'Pieces

        ws.Cells(LR, "H").Value = lstRejects.List(i, 3) 'Disposition
    Next i

End Sub
Private Sub CreateReworkTransactions()

    Dim i As Long
    Dim Disposition As String
    Dim ToProcess As String

    For i = 0 To lstRejects.ListCount - 1

        Disposition = Trim(lstRejects.List(i, 3))

        If Left(UCase(Disposition), 10) = "RETURN TO " Then

            ToProcess = Trim(Mid(Disposition, 11))

            CreateRework _
                txtBatch.Value, _
                lblPart.Caption, _
                "Inspection", _
                ToProcess, _
                CDbl(lstRejects.List(i, 1)), _
                lstRejects.List(i, 0)

        End If

    Next i

End Sub
Private Sub UpdateRemaining()

    Dim BatchNo As String
    Dim RemainingWt As Double

    BatchNo = Trim(txtBatch.Value)

    If BatchNo = "" Then Exit Sub

    RemainingWt = RemainingInspectionWeight(BatchNo)

    RemainingWt = RemainingWt _
                  - val(txtAcceptWeight.Value) _
                  - TotalRejectWeight()

    lblRemaining.Caption = _
        Format(RemainingWt, "0.00")

End Sub
Private Function TotalRejectWeight() As Double

    Dim i As Long

    For i = 0 To lstRejects.ListCount - 1

        TotalRejectWeight = _
            TotalRejectWeight + _
            CDbl(lstRejects.List(i, 1))

    Next i

End Function
Private Sub cmdSave_Click()

    Dim AcceptWt As Double
    Dim RejectWt As Double

    Dim ws As Worksheet
    Dim LR As Long

    Dim InspBaseWt As Double

    AcceptWt = val(txtAcceptWeight.Value)

    RejectWt = TotalRejectWeight()

    If AcceptWt = 0 Then

        MsgBox "Enter Accept Weight"

        Exit Sub

    End If

   ' Check what actually happened to THIS bag, not just whether
    ' Trim is nominally required for the part - a bag can get
    ' trimmed anyway via the "Trimming Not Required - confirm
    ' anyway?" override, and once that's happened, real material
    ' has genuinely been removed regardless of the Part Master flag.
    Dim ActualTrimmedWt As Double
    ActualTrimmedWt = GetTrimmedWeight(txtBatch.Value)

    If ActualTrimmedWt > 0.001 Then
        InspBaseWt = ActualTrimmedWt
    Else
        InspBaseWt = Getproductionweight(txtBatch.Value)
    End If


    Dim RemBeforeInsp As Double
    Dim TotalThisEntry As Double
    Dim InspTolerance As Double

    RemBeforeInsp = RemainingInspectionWeight(txtBatch.Value)
    TotalThisEntry = AcceptWt + RejectWt
    InspTolerance = GetEntryTolerance(InspBaseWt)

    If TotalThisEntry > RemBeforeInsp + InspTolerance Then

        MsgBox "Entered weight (Accept + Reject = " & Format(TotalThisEntry, "0.000") & _
               " Kg) exceeds remaining (" & Format(RemBeforeInsp, "0.000") & _
               " Kg) by more than the allowed tolerance (" & _
               Format(InspTolerance, "0.000") & " Kg)." & vbCrLf & _
               "Cannot save.", vbExclamation, "Weight Exceeds Remaining"

        Exit Sub

    ElseIf TotalThisEntry > RemBeforeInsp Then

        If MsgBox( _
            "Accept Weight: " & Format(AcceptWt, "0.000") & " Kg" & vbCrLf & _
            "Reject Weight: " & Format(RejectWt, "0.000") & " Kg" & vbCrLf & _
            "Remaining Before This Entry: " & Format(RemBeforeInsp, "0.000") & " Kg" & vbCrLf & _
            "This is " & Format(TotalThisEntry - RemBeforeInsp, "0.000") & _
            " Kg over remaining, within tolerance." & vbCrLf & vbCrLf & _
            "Confirm to proceed?", _
            vbQuestion + vbYesNo, "Confirm Overage") = vbNo Then

            Exit Sub

        End If

    ElseIf TotalThisEntry < RemBeforeInsp Then

        Dim RemAfterInsp As Double
        RemAfterInsp = RemBeforeInsp - TotalThisEntry

        If Not IsWithinTolerance(RemAfterInsp, InspBaseWt) Then

            If MsgBox( _
                "Accept Weight: " & Format(AcceptWt, "0.000") & " Kg" & vbCrLf & _
                "Reject Weight: " & Format(RejectWt, "0.000") & " Kg" & vbCrLf & _
                "Remaining Before This Entry: " & Format(RemBeforeInsp, "0.000") & " Kg" & vbCrLf & _
                "Balance After This Entry: " & Format(RemAfterInsp, "0.000") & " Kg" & vbCrLf & vbCrLf & _
                "This is a PARTIAL inspection entry - it does not finish this bag." & vbCrLf & _
                "Confirm to proceed?", _
                vbQuestion + vbYesNo, "Confirm Partial Entry") = vbNo Then

                Exit Sub

            End If

        End If
        ' Else: leftover is already within tolerance - no separate
        ' "partial" warning here. CheckAndUpdateBagStatus (after the
        ' save) asks the single "mark as complete?" confirmation instead.

    End If

    Dim OutstandingReworkInsp As Double
    Dim InspLooksLikeRework As Boolean
    Dim InspEntryType As String

    OutstandingReworkInsp = GetTotalOutstandingRework( _
        txtBatch.Value, "Inspection", InspBaseWt, "INSPECTION_LOG")

    InspLooksLikeRework = (OutstandingReworkInsp > 0.001)

    If chkRework.Value <> InspLooksLikeRework Then

        If MsgBox( _
            "Rework checkbox is " & IIf(chkRework.Value, "CHECKED", "UNCHECKED") & _
            ", but the Rejection Log shows " & Format(OutstandingReworkInsp, "0.000") & _
            " Kg of outstanding rework routed to Inspection for this bag." & vbCrLf & vbCrLf & _
            "Continue anyway with your checkbox setting?", _
            vbQuestion + vbYesNo, "Check Rework Setting") = vbNo Then

            Exit Sub

        End If

    End If

    If chkRework.Value Then
        InspEntryType = "REWORK"
    Else
        InspEntryType = "ORIGINAL"
    End If

    Dim ConfirmMsg As String
    ConfirmMsg = "Confirm Inspection Entry:" & vbCrLf & vbCrLf & _
        "Batch: " & txtBatch.Value & vbCrLf & _
        "Part: " & lblPart.Caption & vbCrLf & _
        "Operator: " & cboOperator.Value & vbCrLf & vbCrLf & _
        "Accept Weight: " & Format(AcceptWt, "0.000") & " Kg (" & lblAcceptPieces.Caption & " pcs)" & vbCrLf & _
        "Reject Weight: " & Format(RejectWt, "0.000") & " Kg (" & lstRejects.ListCount & " reason(s))" & vbCrLf & _
        "Entry Type: " & InspEntryType & vbCrLf & vbCrLf & _
        "Save this entry?"

    If MsgBox(ConfirmMsg, vbQuestion + vbYesNo, "Confirm Inspection Entry") = vbNo Then

        Exit Sub

    End If

    Call SaveRejectLines

    Set ws = Sheets("INSPECTION_LOG")
    If ws.FilterMode Then ws.ShowAllData

    LR = ws.Cells(ws.Rows.Count, "A").End(xlUp).Row + 1

    ws.Cells(LR, "A").Value = Date
    ws.Cells(LR, "B").Value = txtBatch.Value
    ws.Cells(LR, "C").Value = lblPart.Caption
    ws.Cells(LR, "D").Value = cboOperator.Value
    ws.Cells(LR, "E").Value = AcceptWt
    ws.Cells(LR, "F").Value = val(Replace(lblAcceptPieces.Caption, ",", ""))
    ws.Cells(LR, "G").Value = RejectWt

    ws.Cells(LR, "H").Value = _
        Round(RejectWt / GetPartWeight(lblPart.Caption), 0)

    ws.Cells(LR, "I").Value = InspEntryType

    Call CreateReworkTransactions

    Call CheckAndUpdateBagStatus( _
        txtBatch.Value, _
        RemainingInspectionWeight(txtBatch.Value), _
        InspBaseWt, _
        "INSPECTED", "Inspected")

    MsgBox "Inspection Saved"

    Call ClearForm

End Sub

Function GetRejectDisposition(RejectReason As String) As String

    Dim ws As Worksheet
    Dim LR As Long
    Dim r As Long

    Set ws = Sheets("REJECT_MASTER")
    If ws.FilterMode Then ws.ShowAllData

    LR = ws.Cells(ws.Rows.Count, "B").End(xlUp).Row

    For r = 2 To LR

        If Trim(ws.Cells(r, "B").Value) = Trim(RejectReason) Then

            GetRejectDisposition = ws.Cells(r, "C").Value
            Exit Function

        End If

    Next r

End Function

Private Sub UserForm_KeyDown(ByVal KeyCode As MSForms.ReturnInteger, ByVal Shift As Integer)

    If KeyCode = vbKeyReturn Then

        Dim skipAdvance As Boolean
        skipAdvance = False

        On Error Resume Next
        If TypeOf Me.ActiveControl Is MSForms.TextBox Then
            If Me.ActiveControl.Multiline Then skipAdvance = True
        End If
        On Error GoTo 0

        If Not skipAdvance Then
            KeyCode = 0
            Call MoveToNextControl(Me)
        End If

    End If

End Sub
